== du

[source,bash]
----
# メガバイト単位で降順表示
du -sm ./* | sort -k1 -nr

# 人間が読みやすいサイズ表示で降順ソート
du -sh ./* | sort -h -r
----

du::
+
--
[horizontal]
`-s`::: サマリ表示（合計サイズのみ）
`-m`::: メガバイト単位
`-h`::: 人間が読みやすい単位で表示（K/M/G など）
--

sort::
+
--
[horizontal]
`-k1`::: 第1フィールドでソート
`-n`::: 数値としてソート
`-h`::: 人間が読みやすいサイズ表記を認識してソート
`-r`::: 降順（大きい順）
--
